# A persistent marker lets Companion identify saves made with ARES enabled.
# Country flags are stored in the save. The weekly hook also tags human countries
# when this mod is enabled for an existing campaign that is loaded from a save.
on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = {
					is_ai = no
					NOT = { has_country_flag = ares_ai_companion_v1 }
				}
				set_country_flag = ares_ai_companion_v1
			}
		}
	}
	on_weekly = {
		effect = {
			if = {
				limit = {
					is_ai = no
					NOT = { has_country_flag = ares_ai_companion_v1 }
				}
				set_country_flag = ares_ai_companion_v1
			}
		}
	}
}
